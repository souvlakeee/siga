# Commands the PreToolUse hook wraps with siga. One per line: [pipe|pty] <python regex>.
# The regex is searched in each command segment, from where the command starts
# (after env assignments and sudo/time/nice), so anchor with ^.
#   pipe  run in Claude's own shell, output piped through `siga pipe` (default)
#   pty   run on a pseudo-terminal (`siga run --pty`): for tools that only draw progress there
# Add your own in ~/.config/siga/wrap.txt, same format; they are appended to these.

# JavaScript
pipe ^(npm|pnpm|yarn|bun)\s+(run\s+)?(build|test|lint|typecheck|e2e|render|export|ci)\b
pty  ^(npm|pnpm|yarn|bun)\s+(install|i|add)\b
pipe ^npx\s+(tsc|vite\s+build|next\s+build|remotion\s+render|playwright\s+test|jest|vitest)\b
pipe ^(vite|next)\s+build\b
pipe ^(npx\s+)?remotion\s+render\b

# Rust, Go, Swift, JVM, C
pty  ^cargo\s+(build|test|check|clippy|install|run)\b
pipe ^go\s+(build|test|install)\b
pipe ^make\b
pipe ^xcodebuild\b
pipe ^swift\s+build\b
pipe ^(\./)?gradlew?\s

# Python
pipe ^pytest\b
pipe ^python3?\s+-m\s+pytest\b
pty  ^(pip3?|uv\s+pip)\s+install\b
pty  ^uv\s+(sync|add)\b

# containers
pipe ^docker\s+(build|buildx\s+build|compose\s+(build|up\s+--build))\b
pipe ^docker\s+(pull|push)\b

# transfers and media
pipe ^curl\b(?!.*(-o|--output)\s*/dev/null).*\s(-o|-O|--output|--remote-name)\b
pipe ^wget\b
pipe ^rsync\b
pipe ^scp\b
pipe ^git\s+(clone|fetch|pull)\b
pipe ^yt-dlp\b
pipe ^pv\b
pipe ^ffmpeg\b

# system
pty  ^brew\s+(install|upgrade|reinstall)\b
pipe ^mise\s+run\s+(build|test)\b
